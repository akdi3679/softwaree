## OBJECTIVE
Calculate the actual RPO: how stale is the data we'd lose if the disk died right now?

## REQUIRED IMPLEMENTATION

Create `product/apps/admin/src-tauri/src/backup/rpo.rs`:

```rust
use chrono::{DateTime, Utc};
use serde::Serialize;
use sqlx::SqlitePool;

use crate::error::AppResult;

#[derive(Debug, Serialize)]
pub struct RpMeasurement {
    pub project_id: String,
    pub last_backup_at: Option<String>,
    pub last_event_at: String,
    pub rpo_seconds: i64,
    pub rto_estimate_seconds: i64,
    pub status: RpoStatus,
}

#[derive(Debug, Serialize, PartialEq, Eq)]
#[serde(rename_all = "snake_case")]
pub enum RpoStatus {
    Ok,           // last backup < 24h ago
    Warning,      // last backup 24-48h ago
    Critical,     // last backup > 48h ago or never
}

const RPO_OK_SECS: i64 = 24 * 3600;
const RPO_WARN_SECS: i64 = 48 * 3600;
const RTO_ESTIMATE_SECS: i64 = 4 * 3600; // 4 hours

pub async fn measure(pool: &SqlitePool, project_id: &str) -> AppResult<RpMeasurement> {
    let last_backup: Option<(Option<String>,)> = sqlx::query_as(
        "SELECT MAX(created_at) FROM cloud_backups WHERE project_id = ?",
    )
    .bind(project_id)
    .fetch_optional(pool)
    .await?;
    let last_event: (String,) = sqlx::query_as("SELECT MAX(occurred_at) FROM events")
        .fetch_one(pool)
        .await?;

    let now = Utc::now();
    let rpo_seconds = last_backup
        .and_then(|(s,)| s)
        .and_then(|s| DateTime::parse_from_rfc3339(&s).ok())
        .map(|d| (now - d.with_timezone(&Utc)).num_seconds())
        .unwrap_or(i64::MAX);

    let status = match rpo_seconds {
        x if x <= RPO_OK_SECS => RpoStatus::Ok,
        x if x <= RPO_WARN_SECS => RpoStatus::Warning,
        _ => RpoStatus::Critical,
    };

    Ok(RpMeasurement {
        project_id: project_id.to_string(),
        last_backup_at: last_backup.and_then(|(s,)| s),
        last_event_at: last_event.0,
        rpo_seconds: if rpo_seconds == i64::MAX { -1 } else { rpo_seconds },
        rto_estimate_seconds: RTO_ESTIMATE_SECS,
        status,
    })
}
```
