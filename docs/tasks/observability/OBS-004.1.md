## REQUIRED IMPLEMENTATION

Add to `Cargo.toml`:
```toml
opentelemetry = "0.22"

tracing = "0.1"
```

Create `product/apps/admin/src-tauri/src/observability/trace.rs`:

```rust
use tracing::{Span, instrument};
use tracing_opentelemetry::OpenTelemetrySpanExt;

/// Instrument a command: opens a span with the command ID, sets OTEL context.
#[instrument(skip(payload), fields(command_id = %cmd.id, command_type = %cmd.command_type, project_id = %cmd.project_id))]
pub fn trace_command(cmd: &crate::commands::CommandEnvelope<serde_json::Value>, payload: &serde_json::Value) -> Span {
    tracing::info!(target: "command", "received {} on project {}", cmd.command_type, cmd.project_id);
    Span::current()
}

/// Add the current trace context to a sync frame so the receiver can continue the trace.
pub fn current_trace_context() -> Vec<u8> {
    use opentelemetry::global;
    use std::collections::HashMap;
    let cx = tracing::Span::current().context();
    let mut propagator = opentelemetry_sdk::propagation::TraceContextPropagator::new();
    let mut carrier: HashMap<String, String> = HashMap::new();
    propagator.inject_context(&cx, &mut opentelemetry::propagation::Injector::new(&mut carrier));
    serde_json::to_vec(&carrier).unwrap_or_default()
}
```
