```rust
use prometheus::{Counter, Encoder, Histogram, HistogramOpts, Opts, Registry, TextEncoder};
use std::sync::Arc;
use std::net::SocketAddr;
use tokio::net::TcpListener;
use tokio::io::AsyncWriteExt;

pub struct Metrics {
    pub registry: Registry,
    pub commands_total: Counter,
    pub events_appended_total: Counter,
    pub sync_clients_active: prometheus::Gauge,
    pub module_load_duration: Histogram,
    pub http_request_duration: Histogram,
}

impl Metrics {
    pub fn new() -> Self {
        let registry = Registry::new();
        let commands_total = Counter::with_opts(Opts::new("admin_commands_total", "Total commands processed")).unwrap();
        let events_appended_total = Counter::with_opts(Opts::new("admin_events_appended_total", "Total events appended")).unwrap();
        let sync_clients_active = prometheus::Gauge::with_opts(Opts::new("admin_sync_clients_active", "Active sync clients")).unwrap();
        let module_load_duration = Histogram::with_opts(
            HistogramOpts::new("admin_module_load_seconds", "Module load time")
                .buckets(vec![0.001, 0.01, 0.1, 1.0, 10.0])
        ).unwrap();
        let http_request_duration = Histogram::with_opts(
            HistogramOpts::new("admin_http_request_seconds", "HTTP request duration")
                .buckets(vec![0.001, 0.01, 0.1, 1.0, 10.0])
        ).unwrap();
        registry.register(Box::new(commands_total.clone())).unwrap();
        registry.register(Box::new(events_appended_total.clone())).unwrap();
        registry.register(Box::new(sync_clients_active.clone())).unwrap();
        registry.register(Box::new(module_load_duration.clone())).unwrap();
        registry.register(Box::new(http_request_duration.clone())).unwrap();
        Self { registry, commands_total, events_appended_total, sync_clients_active, module_load_duration, http_request_duration }
    }
}

/// Start the metrics HTTP server on the given address.
pub async fn serve(metrics: Arc<Metrics>, addr: SocketAddr) -> std::io::Result<()> {
    let listener = TcpListener::bind(addr).await?;
    tracing::info!(%addr, "metrics server listening");
    loop {
        let (mut socket, _peer) = listener.accept().await?;
        let metrics = metrics.clone();
        tokio::spawn(async move {
            // Read request (we don't actually inspect it)
            let mut buf = [0u8; 4096];
            let _ = tokio::time::timeout(std::time::Duration::from_secs(2), socket.readable()).await;
            let _ = socket.try_read(&mut buf);

            // Render
            let mut buffer = Vec::new();
            let encoder = TextEncoder::new();
            let metric_families = metrics.registry.gather();
            encoder.encode(&metric_families, &mut buffer).unwrap();

            // Respond
            let response = format!(
                "HTTP/1.1 200 OK\r\ncontent-type: text/plain; version=0.0.4\r\ncontent-length: {}\r\nconnection: close\r\n\r\n",
                buffer.len()
            );
            let _ = socket.write_all(response.as_bytes()).await;
            let _ = socket.write_all(&buffer).await;
            let _ = socket.shutdown().await;
        });
    }
}
```

```rust
.setup(|app| {
    let metrics = Arc::new(crate::observability::metrics::Metrics::new());
    app.handle().manage(metrics.clone());
    // Spawn metrics server on tailnet
    let metrics_addr: SocketAddr = "100.100.100.50:9090".parse().unwrap_or_else(|_| "0.0.0.0:9090".parse().unwrap());
    let metrics_clone = metrics.clone();
    tokio::spawn(async move {
        if let Err(e) = crate::observability::metrics::serve(metrics_clone, metrics_addr).await {
            tracing::error!(error = %e, "metrics server failed");
        }
    });
    Ok(())
})
```
