## OBJECTIVE
Alerts for common issues.

## REQUIRED IMPLEMENTATION

Create `internal-infra/grafana/alerts.yml`:

```yaml
groups:
  - name: product-cloud
    interval: 30s
    rules:
      - alert: CloudAPIHigh5xxRate
        expr: rate(http_requests_total{status=~"5.."}[5m]) > 0.01
        for: 5m
        labels:
          severity: critical
        annotations:
          summary: "Cloud API 5xx > 1%"
          description: "{{ $value }} of 5xx on {{ $labels.route }}"

      - alert: AdminSyncQueueBacklog
        expr: sync_outbox_pending > 100
        for: 10m
        labels:
          severity: warning
        annotations:
          summary: "Outbox backlog > 100"

      - alert: BackupFailed
        expr: rate(backup_failures_total[1h]) > 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "Backup failed for {{ $labels.project_id }}"

      - alert: UserDisconnected
        expr: user_last_seen_minutes > 30
        for: 30m
        labels:
          severity: info
        annotations:
          summary: "User offline for 30+ min"

      - alert: CloudCertExpiresSoon
        expr: ssl_cert_expiry_days < 30
        for: 1h
        labels:
          severity: warning
        annotations:
          summary: "TLS cert expires in {{ $value }} days"

      - alert: PostgresConnectionsHigh
        expr: pg_stat_activity_count > 80
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Postgres connections > 80"
```
