```rust
use chrono::Utc;
use serde::Serialize;
use std::sync::Arc;
use std::time::Duration;
use tokio::sync::mpsc;
use tokio::sync::Mutex;

#[derive(Debug, Clone, Serialize)]
pub struct LogLine {
    pub timestamp: String,
    pub level: String,
    pub target: String,
    pub message: String,
    pub fields: serde_json::Value,
}

pub struct LogShipper {
    tx: mpsc::Sender<LogLine>,
}

impl LogShipper {
    /// Start a log shipper. Logs are batched (10 lines or 1 second) and POSTed to the Cloud.
    pub fn start(cloud_url: String, auth_token: Arc<Mutex<String>>) -> Self {
        let (tx, mut rx) = mpsc::channel::<LogLine>(1000);
        tokio::spawn(async move {
            let mut batch: Vec<LogLine> = vec![];
            let mut interval = tokio::time::interval(Duration::from_secs(1));
            interval.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
            loop {
                tokio::select! {
                    Some(line) = rx.recv() => {
                        batch.push(line);
                        if batch.len() >= 10 {
                            flush(&cloud_url, &auth_token, &mut batch).await;
                        }
                    }
                    _ = interval.tick() => {
                        if !batch.is_empty() {
                            flush(&cloud_url, &auth_token, &mut batch).await;
                        }
                    }
                }
            }
        });
        Self { tx }
    }

    /// Submit a log line. Non-blocking.
    pub async fn submit(&self, line: LogLine) {
        let _ = self.tx.send(line).await;
    }
}

async fn flush(cloud_url: &str, auth_token: &Arc<Mutex<String>>, batch: &mut Vec<LogLine>) {
    if batch.is_empty() { return; }
    let token = auth_token.lock().await.clone();
    let url = format!("{cloud_url}/v1/logs/ingest");
    let body = serde_json::json!({ "logs": &*batch });
    let client = reqwest::Client::new();
    let result = client.post(&url)
        .bearer_auth(&token)
        .json(&body)
        .send().await;
    if let Err(e) = result {
        tracing::warn!(error = %e, "log ship failed");
    }
    batch.clear();
}

/// Submit a log line (helper for use anywhere in the code).
pub async fn log(shipper: &LogShipper, level: &str, target: &str, message: &str, fields: serde_json::Value) {
    shipper.submit(LogLine {
        timestamp: Utc::now().to_rfc3339(),
        level: level.to_string(),
        target: target.to_string(),
        message: message.to_string(),
        fields,
    }).await;
}
```

Create `platform-cloud/src/log/ingest.ts`:

```typescript
import { Hono } from 'hono';
import { z } from 'zod';

const LogBatchSchema = z.object({
  logs: z.array(z.object({
    timestamp: z.string().datetime(),
    level: z.string(),
    target: z.string(),
    message: z.string(),
    fields: z.unknown(),
  })),
});

export const logIngestRoutes = new Hono()
  .post('/v1/logs/ingest', async (c) => {
    const authAccountId = c.get('accountId');
    if (!authAccountId) return c.json({ error: 'unauthorized' }, 401);
    const body = LogBatchSchema.parse(await c.req.json());
    // Forward to Loki if configured
    const lokiUrl = process.env.LOKI_URL;
    if (lokiUrl) {
      const streams = [{
        stream: { account_id: authAccountId, job: 'admin' },
        values: body.logs.map((l) => [
          String(new Date(l.timestamp).getTime() * 1_000_000), // nanoseconds
          JSON.stringify({ level: l.level, target: l.target, message: l.message, ...l.fields }),
        ]),
      }];
      const r = await fetch(`${lokiUrl}/loki/api/v1/push`, {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ streams }),
      });
      if (!r.ok) {
        return c.json({ error: 'loki_push_failed', status: r.status }, 502);
      }
    }
    return c.json({ accepted: body.logs.length });
  });
```
